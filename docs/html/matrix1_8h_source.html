570template <typename T>
571std::vector<T> Matrix1<T>::GaussElim(Matrix1<T> &matrix, size_t &sign, bool reduced) const {
572 std::vector<T> pivots;
573 size_t pivot_row = 0;
574
575 for (size_t col = 0; col < matrix.m_cols && pivot_row < matrix.m_rows; ++col) {
576
577 // Buscar pivote
578 size_t found = matrix.m_rows;
579 for (size_t row = pivot_row; row < matrix.m_rows; ++row)
580 if (matrix.m_pMat[row][col] != T{}) { found = row; break; }
581 if (found == matrix.m_rows) { pivots.push_back(T{0}); continue; }
582
583 // Intercambiar filas
584 if (found != pivot_row) {
585 swap(matrix.m_pMat[found], matrix.m_pMat[pivot_row]);
586 sign *= -1;
587 }
588
589 T pivot = matrix.m_pMat[pivot_row][col];
590 pivots.push_back(pivot);
591
592 // Normalizar fila pivote
593 for (size_t j = 0; j < matrix.m_cols; ++j)
594 matrix.m_pMat[pivot_row][j] /= pivot;
595
596 // Eliminar filas - paralelizado con std::thread
597 size_t start = reduced ? 0 : pivot_row + 1;
598
599 std::vector<std::thread> threads;
600 for (size_t row = start; row < matrix.m_rows; ++row) {
601 if (row == pivot_row) continue;
602 threads.emplace_back([&matrix, col, row, pivot_row]() {
603 T factor = matrix.m_pMat[row][col];
604 for (size_t j = 0; j < matrix.m_cols; ++j)
605 matrix.m_pMat[row][j] -= factor * matrix.m_pMat[pivot_row][j];
606 });
607 }
608 for (auto &t : threads) t.join();
609
610 ++pivot_row;
611 }
612 return pivots;
613}
